TYPES

type ConnResponseError struct {
	crsadapter.ConnResponse
}
    ConnResponseError is an error type wrapping a ConnResponse.

func (err ConnResponseError) Error() string
    Error returns err's underlying string, which may be the empty string.

func (err ConnResponseError) Is(target error) bool
    Is indicates whether err matches target. If target is nil and err's
    underlying string is nil, Is returns true. In general, it is preferable to
    use [ConnResponse.ToError] than wrap ConnResponse in a ConnResponseError
    directly.

type Server struct {
	// Has unexported fields.
}
    Server serves requests from adapters and other such things.

func New(
	sl *slog.Logger,
	lPath string,
	iob *iobroker.Broker,
) (*Server, error)
    New listens on the unix socket lPath, unlinking it if it exists, and returns
    a new Server ready for a call to Run.

func (s *Server) Addr() net.Addr
    Addr returns s's listener's address.

func (s *Server) Run(
	ctx context.Context,
) error
    Run handles adapter (and such) clients.
